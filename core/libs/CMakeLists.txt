function(anybox_static_runtime target)
    if(MINGW)
        target_link_options(${target} PRIVATE -static)
    endif()
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -fmacro-prefix-map=${CMAKE_SOURCE_DIR}/=)
    endif()
endfunction()

function(anybox_add_xdll name)
    cmake_parse_arguments(ARG "" "" "SOURCES;LINK" ${ARGN})
    set(target ab_${name})
    set(sources ${ARG_SOURCES} ${ANYBOX_REGISTRATION_SOURCE})
    set(forwards ${CMAKE_CURRENT_SOURCE_DIR}/Forwards.def)
    if(ANYBOX_ORDINALS_${name})
        set(pinned ${ANYBOX_ORDINALS_${name}})
        set(def_lines "EXPORTS\n")
        if(EXISTS ${forwards})
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${forwards})
            file(STRINGS ${forwards} forward_lines)
            foreach(line IN LISTS forward_lines)
                string(STRIP "${line}" line)
                if(line STREQUAL "" OR line STREQUAL "EXPORTS")
                    continue()
                endif()
                string(REGEX MATCH "^[A-Za-z_][A-Za-z0-9_]*" export_name "${line}")
                foreach(entry IN LISTS pinned)
                    if(entry MATCHES "^${export_name}=([0-9]+)$")
                        string(APPEND line " @${CMAKE_MATCH_1}")
                        list(REMOVE_ITEM pinned ${entry})
                    endif()
                endforeach()
                string(APPEND def_lines "    ${line}\n")
            endforeach()
        endif()
        foreach(entry IN LISTS pinned)
            string(REPLACE "=" " @" entry "${entry}")
            string(APPEND def_lines "    ${entry}\n")
        endforeach()
        file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${target}.def CONTENT "${def_lines}" @ONLY)
        list(APPEND sources ${CMAKE_CURRENT_BINARY_DIR}/${target}.def)
    elseif(EXISTS ${forwards})
        list(APPEND sources ${forwards})
    endif()
    add_library(${target} SHARED ${sources})
    set_target_properties(${target} PROPERTIES PREFIX "" OUTPUT_NAME ${target}
        RUNTIME_OUTPUT_DIRECTORY ${ANYBOX_LIBS_DIR} ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
    target_compile_definitions(${target} PRIVATE ABX_LIBRARY_NAME="${name}")
    target_include_directories(${target} PRIVATE ${ANYBOX_LIBS_INCLUDE})
    target_link_libraries(${target} PRIVATE abx_runtime ${ARG_LINK})
    anybox_static_runtime(${target})
    add_dependencies(libs ${target})
endfunction()
